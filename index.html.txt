/* FAVORITES */

let favorites=[];

try{
favorites=
JSON.parse(
localStorage.getItem("kimediFavorites")||"[]"
);
}catch(e){
favorites=[];
}

function saveFavorites(){

localStorage.setItem(
"kimediFavorites",
JSON.stringify(favorites)
);

updateFavoriteCount();

}

function updateFavoriteCount(){

document.getElementById("favoriteCount")
.innerText=favorites.length;

}

function isFavorite(type,index){

return favorites.some(
x=>x.type===type&&x.index===index
);

}

function toggleFavorite(type,index,button){

const existing=
favorites.findIndex(
x=>x.type===type&&x.index===index
);

if(existing>=0){

favorites.splice(existing,1);

button.classList.remove("liked");

button.innerHTML="♡";

showToast("از علاقه‌مندی‌ها حذف شد");

}else{

favorites.push({
type:type,
index:index
});

button.classList.add("liked");

button.innerHTML="♥";

const heart=document.createElement("div");

heart.className="heartParticle";

heart.innerText="♥";

const rect=button.getBoundingClientRect();

heart.style.left=
rect.left+rect.width/2+"px";

heart.style.top=
rect.top+rect.height/2+"px";

document.body.appendChild(heart);

setTimeout(()=>{
heart.remove();
},900);

showToast("❤️ به علاقه‌مندی‌ها اضافه شد");

}

saveFavorites();

}

function showPage(page){

document
.querySelectorAll(".page")
.forEach(
x=>x.classList.remove("active")
);

const target=
document.getElementById(page);

if(target){
target.classList.add("active");
}

.querySelectorAll(".nav button")
.forEach(btn=>{

"active",
btn.dataset.page===page

});

}

}

/* FILTER */

function filterProducts(type,filter,button){

button.parentElement
.querySelectorAll(".filter")
.forEach(
x=>x.classList.remove("active")
);

button.classList.add("active");

let data=products[type];

if(filter==="available"){
data=data.filter(p=>p[3]);
}

if(filter==="off"){
data=data.filter(p=>!p[3]);
}

renderProducts(type,data);

}
